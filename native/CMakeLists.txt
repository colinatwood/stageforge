cmake_minimum_required(VERSION 3.20)
project(stageforge_devices LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(stageforge_devices device_monitor.cpp device_identity.cpp)
target_include_directories(stageforge_devices PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_compile_definitions(stageforge_devices PRIVATE NOMINMAX)
  target_link_libraries(stageforge_devices PUBLIC ole32 uuid bcrypt)
elseif(APPLE)
  target_link_libraries(stageforge_devices PUBLIC "-framework CoreAudio" "-framework CoreFoundation")
else()
  message(FATAL_ERROR "Device monitor requires Windows or macOS")
endif()
add_executable(device_lifecycle_smoke device_lifecycle_smoke.cpp)
target_link_libraries(device_lifecycle_smoke PRIVATE stageforge_devices)
enable_testing()
add_test(NAME device_lifecycle COMMAND device_lifecycle_smoke)
set_tests_properties(device_lifecycle PROPERTIES TIMEOUT 30)
if(APPLE)
  add_executable(coreaudio_device_events coreaudio_device_events.cpp)
  target_link_libraries(coreaudio_device_events PRIVATE stageforge_devices)
  add_test(NAME coreaudio_device_events COMMAND coreaudio_device_events)
  set_tests_properties(coreaudio_device_events PROPERTIES TIMEOUT 45)
endif()
